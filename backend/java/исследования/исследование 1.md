package com.ths.product.service;

import com.ths.product.entity.Product;
import com.ths.product.producers.ProductEventProducer;
import com.ths.product.repository.ProductRepository;
import com.ths.product.repository.ProductStatusRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
public class ProductService {

    private final ProductRepository productRepository;
    private final ProductStatusRepository productStatusRepository;
    private final ProductEventProducer productEvent;

    public boolean productExists(Long id) {
        return productRepository.existsById(id);
    }

    public List<Product> getProductsByIds(List<Long> productIds) {
        return productRepository.findAllById(productIds);
    }

    public Optional<Product> getProductById(Long id) {
        return productRepository.findById(id);
    }

    @Transactional
    public void discontinueProduct(Long productId) {
        var id = productStatusRepository.findByCode("discontinued")
                .orElseThrow()
                .getId();

        var product = productRepository.findById(productId)
                .orElseThrow();

        product.setStatusId(id);

        productEvent.sendProductDiscontinued(productId);
    }
}

- Корректно ли возвращать Optional из этого метода? И если нет, то как в индустрии используют альтернативы?

Выводы:

- Альтернатива по сути единственная - выброс исключения. И его тогда можно было бы поймать в глобальном обработчике и не проверять на ifPresent в контроллере.
  - Но это было бы говно-кодом по сути, потому что исключение использовалось бы не по назначению.
    - Основное назначение исключения - выразить ситуацию, которая с точки зрения программы является ошибочной, некорректной (как с технической, так и с бизнесовой, не важно). Например, пытаться задать человеку отрицательный вес.
- Так что Optional - это нормально.
